# robots.txt — MUMU Solutions
# Updated: 2026-09-21
#
# Policy: only the homepage is indexable. Everything else is denied by the
# Disallow below and opened back up case by case. /llms.txt and /ads.txt are
# two of those exceptions — a file written for crawlers is useless if
# robots.txt hides it, and AdSense flags the domain while ads.txt is
# unreachable.
#
# Two things must stay allowed for the site to work correctly in search and
# in social previews:
#   * /images/ and /fonts/ — Google renders the page before indexing it. A
#     crawler that cannot fetch the logo, the share image, or the webfonts
#     sees a broken page and judges it accordingly.
#   * the social crawlers below — they fetch og-image.png directly. Blocked,
#     a link shared on WhatsApp or LinkedIn renders as a grey box.

User-agent: *
Allow: /$
Allow: /index.html
Allow: /images/
Allow: /fonts/
Allow: /sitemap.xml
Allow: /llms.txt
Allow: /ads.txt
Disallow: /

# Social crawlers need the Open Graph image, not just the page.
User-agent: facebookexternalhit
Allow: /$
Allow: /index.html
Allow: /images/
Disallow: /

User-agent: Twitterbot
Allow: /$
Allow: /index.html
Allow: /images/
Disallow: /

User-agent: LinkedInBot
Allow: /$
Allow: /index.html
Allow: /images/
Disallow: /

User-agent: WhatsApp
Allow: /$
Allow: /index.html
Allow: /images/
Disallow: /

# AdSense reads the page to decide which ads suit it. The blanket Disallow
# above already leaves the homepage reachable, but Mediapartners-Google obeys
# only its most specific matching group — so spell it out here rather than
# letting the allowance depend on the User-agent: * block staying as it is.
User-agent: Mediapartners-Google
Allow: /$
Allow: /index.html
Allow: /images/
Allow: /fonts/
Disallow: /

Disallow: /

User-agent: DotBot
Disallow: /

User-agent: rogerbot

# Canonical host is the apex — it must match CNAME, <link rel="canonical">,
# og:url and every <loc> in the sitemap, or ranking signal is split in two.

# custom domain configured for the Pages deployment. That redirect is the
# reason no www URL appears anywhere in this repository.
